{{/* Ceci est un override que j'ai ajouté pour faire en sorte que la page d'accueil ait un lien "next page", et qu'il soit à droite, même s'il y a aucun lien "prev" */}} {{/* Porté en août 2026 : "docs/prev-next-cache" a été supprimé de hugo-book, remplacé par "docs/menu-section" + "docs/menu-section-pages". */}} {{- with (partialCached "docs/menu-section" .Site) -}} {{- $pages := partialCached "docs/menu-section-pages" . -}} {{- $allPages := slice $.Site.Home | append $pages -}} {{- $prev := $allPages.Next $ -}} {{- $next := $allPages.Prev $ -}} {{- if or $prev $next }}
{{ with $prev }} {{ partial "docs/title" . }} {{ end }} {{ with $next }} {{ partial "docs/title" . }} {{ end }}
{{ end -}} {{- end -}}
